# hycom_sbda -- pinned dependencies for the diffusion training core.
# Versions captured from the on-prem `datorch` env (Python 3.11+, CUDA 12.6).
# Install:  pip install -r requirements.txt
#
# torch comes from the PyTorch CUDA-12.6 wheel index, matching the on-prem A100s
# and the p4d/p5 instances this targets. Inside aws/Dockerfile the base image
# already ships an arch-matched torch, so the two lines below are STRIPPED at
# build time -- honouring the pin there would overwrite it.
#
# NOTE -- Blackwell GPUs (sm_100/sm_120: B200, RTX PRO ...): cu126 wheels carry
# no kernels for them and every CUDA op fails at runtime. Use the cu128 index:
#   pip install --index-url https://download.pytorch.org/whl/cu128 torch==2.8.0
# Verify with: python -c "import torch; print(torch.cuda.get_arch_list())"
--extra-index-url https://download.pytorch.org/whl/cu126
torch==2.8.0

# -- Core numerics / IO ------------------------------------------------------
# This repo imports exactly four third-party packages: torch, numpy, yaml, zarr.
# The dataset-BUILDER dependencies (xarray, cv2, pyproj, cftime) are deliberately
# absent -- nothing here imports them. Do not add them "just in case".
numpy==2.3.3
zarr==2.18.7        # HARD constraint: data.py/dataset_spec.py target the v2 API
numcodecs==0.13.1
PyYAML==6.0.3

# -- Remote data (S3 zarr streaming) -----------------------------------------
# Optional at import time: dataset_spec imports fsspec lazily, only for a path
# containing "://". Keep s3fs version-matched to fsspec; it pulls a compatible
# aiobotocore transitively.
fsspec==2024.6.1
s3fs==2024.6.1

# -- Cloud plumbing ----------------------------------------------------------
# `aws s3 sync`, used by aws/common.sh for staging in and persisting results out.
# Left unpinned deliberately: it is tooling, not a numerics dependency, so an
# exact pin buys no reproducibility and only risks breaking the image build.
awscli
